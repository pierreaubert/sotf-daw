//! Supporting faces, canonical diagonals and coverage regression checks.

// Rust guideline compliant 2026-02-21
use super::{Neighbors, spatial_neighbors};
use sotf_host::sofa::{SofaFile, SourcePosition};

fn position(v: [f64; 3]) -> SourcePosition {
    SourcePosition::new(
        v[1].atan2(v[0]).to_degrees() as f32,
        v[2].atan2(v[0].hypot(v[1])).to_degrees() as f32,
        1.0,
    )
}

fn fixture(positions: Vec<SourcePosition>) -> SofaFile {
    let count = positions.len();
    SofaFile {
        sample_rate: 48_000.0,
        num_measurements: count,
        ir_length: 1,
        positions,
        impulse_responses: vec![1.0; 2 * count],
        convention: "SimpleFreeFieldHRIR".into(),
        data_sample_rate: Some(48_000.0),
    }
}

fn cube() -> SofaFile {
    let mut positions = Vec::new();
    for x in [-1.0, 1.0] {
        for y in [-1.0, 1.0] {
            for z in [-1.0, 1.0] {
                positions.push(position([x, y, z]));
            }
        }
    }
    // Explicit poles keep this synthetic dataset's elevation coverage full.
    positions.push(position([0.0, 0.0, 1.0]));
    positions.push(position([0.0, 0.0, -1.0]));
    fixture(positions)
}

fn field((near, weights): Neighbors) -> f64 {
    // An arbitrary non-affine vertex field detects diagonal switches that a
    // constant/linear field alone would conceal.
    near.into_iter()
        .zip(weights)
        .map(|((id, _), w)| (id * id + 1) as f64 * f64::from(w))
        .sum()
}

#[test]
fn exact_nodes_and_constant_fields_survive_all_cube_faces() {
    let sofa = cube();
    for (id, target) in sofa.positions.iter().enumerate() {
        let (near, w) = spatial_neighbors(&sofa, target).unwrap();
        assert_eq!(near[0].0, id);
        assert_eq!(w, [1.0, 0.0, 0.0]);
    }
    for axis in 0..3 {
        for sign in [-1.0, 1.0] {
            for a in [-0.9, -0.3, 0.1, 0.7] {
                for b in [-0.8, -0.1, 0.3, 0.9] {
                    let mut v = [0.0; 3];
                    v[axis] = sign;
                    v[(axis + 1) % 3] = a;
                    v[(axis + 2) % 3] = b;
                    let (_, w) = spatial_neighbors(&sofa, &position(v)).unwrap();
                    assert!(w.iter().all(|w| *w >= 0.0 && w.is_finite()));
                    assert!((w.iter().sum::<f32>() - 1.0).abs() < 2e-6);
                    assert!(
                        w.iter().filter(|&&w| w > 0.0).count() >= 2,
                        "unexpected nearest fallback {v:?}"
                    );
                }
            }
        }
    }
}

#[test]
fn shared_edges_and_coplanar_diagonals_are_continuous() {
    let sofa = cube();
    let mut maximum = 0.0_f64;
    for axis in 0..3 {
        for sign in [-1.0, 1.0] {
            for t in [-0.75, -0.25, 0.0, 0.25, 0.75] {
                for diagonal in [-1.0, 1.0] {
                    let mut a = [0.0; 3];
                    a[axis] = sign;
                    a[(axis + 1) % 3] = t;
                    a[(axis + 2) % 3] = diagonal * t - 1e-7;
                    let mut b = a;
                    b[(axis + 2) % 3] += 2e-7;
                    let difference = (field(spatial_neighbors(&sofa, &position(a)).unwrap())
                        - field(spatial_neighbors(&sofa, &position(b)).unwrap()))
                    .abs();
                    maximum = maximum.max(difference);
                }
                let mut a = [0.0; 3];
                a[axis] = sign;
                a[(axis + 1) % 3] = 1.0 - 1e-7;
                a[(axis + 2) % 3] = t;
                let mut b = a;
                b[(axis + 1) % 3] += 2e-7;
                maximum = maximum.max(
                    (field(spatial_neighbors(&sofa, &position(a)).unwrap())
                        - field(spatial_neighbors(&sofa, &position(b)).unwrap()))
                    .abs(),
                );
            }
        }
    }
    println!("MESH_BOUNDARY maximum_vertex_field_jump={maximum:e}");
    assert!(maximum < 2e-5, "coplanar fan/edge discontinuity {maximum}");
}

#[test]
fn elevation_coverage_and_multiple_radii_have_explicit_fallbacks() {
    let mut sofa = fixture(vec![
        SourcePosition::new(-20.0, -10.0, 1.0),
        SourcePosition::new(20.0, -10.0, 1.0),
        SourcePosition::new(0.0, 20.0, 1.0),
    ]);
    let (_, weights) = spatial_neighbors(&sofa, &SourcePosition::new(0.0, 60.0, 1.0)).unwrap();
    assert_eq!(weights, [1.0, 0.0, 0.0]);
    sofa.positions[1].distance = 2.0;
    assert!(spatial_neighbors(&sofa, &SourcePosition::new(0.0, 0.0, 1.0)).is_none());
}
