"""Offline AAE profile fitting with optional ML, exact verification and fallback.

Input must be a deconvolved impulse response with an explicitly supplied direct
arrival index. Fitting matches finite-window energy decay, not a complex room
field, loudspeaker calibration, overall wet gain, or listener preference.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import time
from pathlib import Path

import numpy as np
from aae_room_features import (
    FEATURE_SPEC,
    LAYOUTS,
    PRESETS,
    Renderer,
    actual_unit,
    describe,
    encode,
    parameters,
)
from aae_room_model import Surrogate
from scipy import optimize
from scipy.io import wavfile
from scipy.stats import qmc

MATCH_RMSE_DB = 3.0  # Fixed engineering profile proxy, not perceptual acceptance.
DEFAULT_VALUES = np.array([1.0, 1.8, 1.2, 0.5, 20.0, 0.6])


class BudgetExhausted(Exception):
    pass


def seed_from_physics(target: dict) -> np.ndarray:
    values = DEFAULT_VALUES.copy()
    fits = target["t20_fits"]
    mid = fits[4]  # wide,125,250,500,1000,2000,4000
    if mid is not None and mid["r_squared"] >= 0.95:
        values[1] = np.clip(mid["rt60_seconds"], 0.3, 6.0)
        for index, fit in [(2, fits[1]), (3, fits[6])]:
            if fit is not None and fit["r_squared"] >= 0.95:
                values[index] = np.clip(
                    fit["rt60_seconds"] / values[1],
                    0.8 if index == 2 else 0.2,
                    2.0 if index == 2 else 1.0,
                )
    p = {
        name: float(value)
        for name, value in zip(
            ("room_size", "rt60", "bass_ratio", "treble_ratio", "pre_delay_ms"),
            values[:5],
            strict=True,
        )
    }
    p["er_level"] = float(values[5])
    return actual_unit(p)


def dictionary_seeds(
    dictionary: Path | None, target: dict, layout: str
) -> list[tuple[np.ndarray, str]]:
    if dictionary is None:
        return []
    data = np.load(dictionary, allow_pickle=False)
    x, y = data["x"], data["y"]
    if (
        x.ndim != 2
        or x.shape[1] != 19
        or y.shape != (len(x), 224)
        or not np.isfinite(x).all()
        or not np.isfinite(y).all()
    ):
        raise ValueError("invalid native seed dictionary")
    selected = x[:, 6 + LAYOUTS.index(layout)] == 1
    if not np.any(selected):
        raise ValueError("native dictionary does not cover requested layout")
    x, y = x[selected], y[selected]
    loss = np.mean(
        (y[:, target["mask"]] - target["features"][target["mask"]]) ** 2, axis=1
    )
    result = []
    for preset in range(len(PRESETS)):
        candidates = np.flatnonzero(x[:, 15 + preset] == 1)
        if len(candidates):
            index = candidates[np.argmin(loss[candidates])]
            result.append((x[index, :6].astype(float), PRESETS[preset]))
    return result


def neural_seeds(
    model: Surrogate, target: dict, layout: str, seeds: list[tuple[np.ndarray, str]]
) -> tuple[list[tuple[np.ndarray, str]], dict]:
    start = time.perf_counter()
    count = 0
    proposed = []
    random = qmc.Sobol(6, scramble=True, seed=20265025).random_base2(6)
    mask = target["mask"]
    desired = target["features"][mask]
    for preset in PRESETS:
        initial = [seed for seed, kind in seeds if kind == preset]
        initial.extend(random)
        encoded = np.array([encode(unit, layout, preset) for unit in initial])
        loss = np.mean((model.predict(encoded)[:, mask] - desired) ** 2, axis=1)
        count += len(initial)
        for index in np.argsort(loss)[:2]:

            def objective(unit, preset=preset):
                nonlocal count
                count += 1
                curve = model.predict(encode(np.clip(unit, 0, 1), layout, preset))
                return float(np.mean((curve[mask] - desired) ** 2))

            fit = optimize.minimize(
                objective,
                np.asarray(initial[index]),
                method="L-BFGS-B",
                bounds=[(0.0, 1.0)] * 6,
                options={"maxiter": 80, "ftol": 1e-9, "eps": 0.0001, "maxls": 12},
            )
            proposed.append((float(fit.fun), np.clip(fit.x, 0, 1), preset))
    proposed.sort(key=lambda row: row[0])
    return [(unit, preset) for _, unit, preset in proposed], {
        "seconds": time.perf_counter() - start,
        "surrogate_evaluations": count,
    }


def fit(
    target: dict,
    renderer: Renderer,
    *,
    layout: str,
    rate: int,
    budget: int = 64,
    model: Surrogate | None = None,
    dictionary: Path | None = None,
) -> dict:
    if budget < 8 or budget > 1024:
        raise ValueError("native fitting budget must be 8..1024")
    mask = target["mask"]
    if np.count_nonzero(mask) < 28:
        raise ValueError("insufficient observed decay times for fitting")
    curves = target["features"].reshape(7, 32)
    coverage = mask.reshape(7, 32)
    informative = sum(
        bool(np.any(curve[valid] <= -25.0) and np.any(curve[valid] >= -5.0))
        for curve, valid in zip(curves, coverage, strict=True)
    )
    wide_observed = curves[0, coverage[0]]
    if informative < 3 or not (
        np.any(wide_observed <= -25.0) and np.any(wide_observed >= -5.0)
    ):
        raise ValueError(
            "insufficient observed decay: need 25 dB coverage in the wideband and at least three analysis curves"
        )
    start = time.perf_counter()
    baseline = dictionary_seeds(dictionary, target, layout)
    physics = seed_from_physics(target)
    default = actual_unit(
        dict(
            zip(
                (
                    "room_size",
                    "rt60",
                    "bass_ratio",
                    "treble_ratio",
                    "pre_delay_ms",
                    "er_level",
                ),
                DEFAULT_VALUES,
                strict=True,
            )
        )
    )
    baseline.extend([(physics, "medium"), (default, "medium")])
    # A dictionary is a strong non-neural baseline using the same native corpus.
    # Without it, retain explicit preset coverage and reproducible global seeds.
    if dictionary is None:
        random = qmc.Sobol(6, scramble=True, seed=20265026).random_base2(3)
        baseline.extend((row, PRESETS[index % 4]) for index, row in enumerate(random))
    proposed = []
    proposal_cost = {"seconds": 0.0, "surrogate_evaluations": 0}
    if model is not None:
        proposed, proposal_cost = neural_seeds(model, target, layout, baseline)
    candidates = proposed + baseline
    cache = {}
    records = []
    best = None

    def evaluate(unit, preset):
        nonlocal best
        unit = np.clip(np.asarray(unit, dtype=float), 0.0, 1.0)
        p = parameters(unit, layout, preset)
        key = json.dumps(p, sort_keys=True, separators=(",", ":"))
        if key in cache:
            return cache[key]
        if len(records) >= budget:
            raise BudgetExhausted
        audio, metadata = renderer.render(unit, layout, preset, rate=rate)
        measured = describe(audio, rate)
        value = float(
            np.mean((measured["features"][mask] - target["features"][mask]) ** 2)
        )
        if not np.isfinite(value):
            raise RuntimeError("native objective is non-finite")
        row = {
            "evaluation": len(records) + 1,
            "mse_db2": value,
            "preset": preset,
            "unit": actual_unit(metadata["params"]).tolist(),
            "params": metadata["params"],
            "native_sha256": metadata["sha256"],
            "native_tail": metadata["native_tail"],
            "native_t20_fits": measured["t20_fits"],
            "per_curve_rmse_db": [
                float(np.sqrt(np.mean((a[valid] - b[valid]) ** 2)))
                for a, b, valid in zip(
                    measured["features"].reshape(7, 32), curves, coverage, strict=True
                )
            ],
            "seconds": time.perf_counter() - start,
        }
        records.append(row)
        cache[key] = value
        if best is None or value < best["mse_db2"]:
            best = row
        return value

    # Evaluate seeds with the exact renderer before admitting any model output.
    for unit, preset in candidates:
        if len(records) >= min(budget // 2, 16):
            break
        evaluate(unit, preset)
    assert best is not None
    seed = np.asarray(best["unit"])
    preset = best["preset"]
    simplex = np.tile(seed, (7, 1))
    for index in range(6):
        simplex[index + 1, index] = np.clip(
            seed[index] + (0.05 if seed[index] <= 0.95 else -0.05), 0, 1
        )
    try:
        optimize.minimize(
            lambda unit: evaluate(unit, preset),
            seed,
            method="Nelder-Mead",
            bounds=[(0.0, 1.0)] * 6,
            options={
                "maxfev": budget * 4,
                "maxiter": budget * 4,
                "initial_simplex": simplex,
                "xatol": 0.0001,
                "fatol": 0.0001,
            },
        )
    except BudgetExhausted:
        pass
    qualified = best["mse_db2"] <= MATCH_RMSE_DB**2
    first = next((row for row in records if row["mse_db2"] <= MATCH_RMSE_DB**2), None)
    return {
        "method": "neural_plus_native" if model is not None else "classical_native",
        "qualified_profile_proxy": qualified,
        "best": best,
        "profile_rmse_db": float(np.sqrt(best["mse_db2"])),
        "native_evaluations": len(records),
        "max_native_evaluations": budget,
        "first_qualified_evaluation": None if first is None else first["evaluation"],
        "fit_seconds": time.perf_counter() - start,
        "proposal_cost": proposal_cost,
        "trace": records,
        "observed_feature_count": int(np.count_nonzero(mask)),
        "informative_decay_curves": informative,
        "match_rmse_limit_db": MATCH_RMSE_DB,
        "measurement_observed_seconds": target["observed_seconds"],
        "measurement_tail_energy_fraction": target[
            "observed_last_tenth_energy_fraction"
        ],
        "status": "profile_match_requires_review"
        if qualified
        else "no_qualified_profile_match",
        "scope": "Finite-window energy profile only; no sound-field, geometry, absolute gain or perceptual match claim.",
    }


def load_ir(path: Path) -> tuple[int, np.ndarray]:
    rate, audio = wavfile.read(path)
    if np.issubdtype(audio.dtype, np.integer):
        if audio.dtype == np.uint8:
            audio = (audio.astype(float) - 128) / 128
        else:
            audio = audio.astype(float) / float(2 ** (np.iinfo(audio.dtype).bits - 1))
    return int(rate), np.asarray(audio, dtype=float)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--ir", type=Path, required=True)
    parser.add_argument("--direct-sample", type=int, required=True)
    parser.add_argument("--renderer", type=Path, required=True)
    parser.add_argument("--temporary", type=Path, required=True)
    parser.add_argument("--layout", choices=LAYOUTS, default="5.1")
    parser.add_argument("--model", type=Path)
    parser.add_argument("--model-sha256")
    parser.add_argument("--dictionary", type=Path)
    parser.add_argument("--budget", type=int, default=64)
    parser.add_argument("--report", type=Path, required=True)
    parser.add_argument("--preset", type=Path)
    args = parser.parse_args()
    source_rate, audio = load_ir(args.ir)
    target = describe(audio, source_rate, direct_sample=args.direct_sample)
    model = None
    model_error = None
    if args.model is not None:
        try:
            if args.model_sha256 is None:
                raise ValueError("explicit model SHA256 is required")
            model = Surrogate(args.model, args.model_sha256)
        except (ValueError, OSError) as error:
            model_error = str(error)
    with Renderer(args.renderer, args.temporary) as renderer:
        classical = fit(
            target,
            renderer,
            layout=args.layout,
            rate=source_rate,
            budget=args.budget,
            dictionary=args.dictionary,
        )
        neural = None
        neural_error = None
        if model is not None:
            try:
                neural = fit(
                    target,
                    renderer,
                    layout=args.layout,
                    rate=source_rate,
                    budget=args.budget,
                    model=model,
                    dictionary=args.dictionary,
                )
            except (ValueError, RuntimeError, FloatingPointError) as error:
                neural_error = str(error)
        # No default replacement: exact native scores choose the output. All
        # model failures and inferior proposals retain the classical result.
        selected = (
            classical
            if neural is None
            or classical["best"]["mse_db2"] <= neural["best"]["mse_db2"]
            else neural
        )
        report = {
            "input_sha256": hashlib.sha256(args.ir.read_bytes()).hexdigest(),
            "input_rate": source_rate,
            "direct_sample": args.direct_sample,
            "feature_spec": FEATURE_SPEC,
            "renderer_sha256": renderer.binary_sha256,
            "model_sha256": None if model is None else model.sha256,
            "model_error": model_error,
            "neural_fit_error": neural_error,
            "classical": classical,
            "neural": neural,
            "selected_method": selected["method"],
            "qualified_profile_proxy": selected["qualified_profile_proxy"],
            "automatic_apply": False,
            "measurement_t20_fits": target["t20_fits"],
        }
    args.report.parent.mkdir(parents=True, exist_ok=True)
    args.report.write_text(json.dumps(report, indent=2, allow_nan=False) + "\n")
    if args.preset is not None and selected["qualified_profile_proxy"]:
        args.preset.write_text(
            json.dumps(selected["best"]["params"], indent=2, allow_nan=False) + "\n"
        )
    print(
        json.dumps(
            {
                "selected_method": report["selected_method"],
                "qualified_profile_proxy": report["qualified_profile_proxy"],
                "model_error": model_error,
                "neural_fit_error": neural_error,
            }
        )
    )


if __name__ == "__main__":
    main()
