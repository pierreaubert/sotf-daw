"""Causal offline key experiments; VAD scores are not sibilance labels."""

import json
import subprocess
from itertools import pairwise
from pathlib import Path

import numpy as np

RATE = 48000
HOP = 480
WINDOW = 960
OPEN_RATIO = 0.35
CLOSE_RATIO = 0.25
LOW_SCORE = 0.2
HIGH_SCORE = 0.8
MAX_AGE_MS = 150.0

PARAMS = {
    "frequency": 6000.0,
    "q": 0.7,
    "threshold": -36.0,
    "ratio": 8.0,
    "attack_ms": 1.0,
    "release_ms": 80.0,
    "mode": "Wideband",
    "mix": 1.0,
    "range_db": 6.0,
    "stereo_link": 1.0,
    "lookahead_ms": 2.0,
    "split_topology": "Minimum-Phase",
    "ms_mode": False,
    "sidechain_external": False,
}


def audio_array(signal):
    x = np.asarray(signal, dtype=np.float32)
    if x.ndim == 1:
        x = x[:, None]
    if (
        x.ndim != 2
        or not 1 <= x.shape[1] <= 8
        or not 1 <= x.size <= 4_000_000
        or not np.isfinite(x).all()
        or np.max(np.abs(x)) > 32
    ):
        raise ValueError("Use bounded finite native frame/channel audio")
    return x


def render(signal, renderer, work, key=None, params=None, rate=RATE, callback=193):
    x = audio_array(signal)
    if key is not None:
        key = audio_array(key)
        if key.shape != x.shape:
            raise ValueError("External key must match programme geometry")
    work = Path(work)
    work.mkdir(parents=True, exist_ok=False)
    x.astype("<f4").tofile(work / "input.f32le")
    settings = dict(PARAMS if params is None else params)
    settings["sidechain_external"] = key is not None
    request = {
        "input": str((work / "input.f32le").resolve()),
        "output": str((work / "output.f32le").resolve()),
        "key": None,
        "sample_rate": rate,
        "channels": x.shape[1],
        "frames": len(x),
        "callback_frames": callback,
        "params": settings,
    }
    if key is not None:
        key.astype("<f4").tofile(work / "key.f32le")
        request["key"] = str((work / "key.f32le").resolve())
    path = work / "request.json"
    path.write_text(json.dumps(request, allow_nan=False) + "\n")
    result = subprocess.run(
        [str(Path(renderer).resolve()), str(path)],
        check=False,
        capture_output=True,
        text=True,
        timeout=120,
    )
    if result.returncode:
        raise ValueError("Native de-esser failed: " + result.stderr)
    metadata = json.loads(result.stdout)
    (work / "metadata.json").write_text(json.dumps(metadata, indent=2) + "\n")
    latency, drained = metadata["latency_samples"], metadata["drained_frames"]
    if (
        type(latency) is not int
        or type(drained) is not int
        or not 0 <= latency <= drained <= 65536
    ):
        raise ValueError("Invalid native de-esser delay/tail")
    path = work / "output.f32le"
    if path.stat().st_size != (len(x) + drained) * x.shape[1] * 4:
        raise ValueError("Invalid native de-esser output size")
    raw = np.fromfile(path, dtype="<f4").reshape(-1, x.shape[1])
    if (
        metadata["status"] != "ok"
        or metadata["input_frames"] != len(x)
        or metadata["output_frames"] != len(raw)
        or metadata["sample_rate"] != rate
        or metadata["channels"] != x.shape[1]
        or not np.isfinite(raw).all()
    ):
        raise ValueError("Invalid native de-esser output")
    return raw[latency : latency + len(x)].copy(), metadata


def observe(signal, observer, model, work):
    x = audio_array(signal)
    if x.shape[1] != 1:
        raise ValueError("Speech observation uses one explicit physical channel")
    work = Path(work)
    work.mkdir(parents=True, exist_ok=False)
    x.astype("<f4").tofile(work / "input.f32le")
    request = {
        "input": str((work / "input.f32le").resolve()),
        "frames": len(x),
        "model": model,
    }
    path = work / "request.json"
    path.write_text(json.dumps(request, allow_nan=False) + "\n")
    result = subprocess.run(
        [str(Path(observer).resolve()), str(path)],
        check=False,
        capture_output=True,
        text=True,
        timeout=300,
    )
    if result.returncode:
        raise ValueError("Speech observer failed: " + result.stderr)
    trace = json.loads(result.stdout)
    if (
        trace["status"] != "ok"
        or trace["sample_rate"] != RATE
        or trace["input_frames"] != len(x)
        or trace["packet_frames"] != HOP
    ):
        raise ValueError("Unsupported speech trace")
    (work / "trace.json").write_text(json.dumps(trace, indent=2) + "\n")
    return trace


def key_policy(signal, method="relative", trace=None):
    """Use only observations received by each packet's start; no EOF padding."""
    x = audio_array(signal)
    if x.shape[1] != 1 or method not in {"relative", "speech-relative"}:
        raise ValueError("Select an explicit mono offline key policy")
    x = x[:, 0]
    rows = [] if trace is None else trace["rows"]
    if any(
        type(row.get("available_at_frame")) is not int or row["available_at_frame"] <= 0
        for row in rows
    ) or any(
        a["available_at_frame"] >= b["available_at_frame"] for a, b in pairwise(rows)
    ):
        raise ValueError("Speech observations must follow strict source chronology")
    keys, decisions = np.empty_like(x), []
    cursor, latest, opened, low_hops, smoothing = 0, None, False, 0, 1.0
    relative_target = 1.0
    alpha = np.exp(-1 / (0.005 * RATE))
    weights = np.hanning(WINDOW)
    frequencies = np.fft.rfftfreq(WINDOW, 1 / RATE)
    band = (frequencies >= 4000) & (frequencies <= 10000)
    for start in range(0, len(x), HOP):
        end = min(start + HOP, len(x))
        while cursor < len(rows) and rows[cursor]["available_at_frame"] <= start:
            row = rows[cursor]
            if row["new_observation"]:
                latest = row
            cursor += 1
        target, reason = relative_target, "relative"
        if method == "speech-relative":
            target, reason = 1.0, "classical_fallback"
            if latest is not None:
                score, age = latest["score"], latest["source_age_ms_at_snapshot"]
                if (
                    score is not None
                    and age is not None
                    and np.isfinite(score)
                    and np.isfinite(age)
                    and 0 <= score <= 1
                    and age >= 0
                    and latest["warmup_complete"]
                ):
                    age = age + (start - latest["available_at_frame"]) * 1000 / RATE
                    if age <= MAX_AGE_MS:
                        if score <= LOW_SCORE:
                            target, reason = 0.0, "low_speech_score"
                        elif score >= HIGH_SCORE:
                            target, reason = relative_target, "speech_context_relative"
        gate = target + (smoothing - target) * alpha ** np.arange(1, end - start + 1)
        keys[start:end] = (x[start:end].astype(np.float64) * gate).astype(np.float32)
        smoothing = float(gate[-1])
        # Complete trailing window becomes usable only in the NEXT packet.
        window = np.zeros(WINDOW)
        available = x[max(0, end - WINDOW) : end]
        window[-len(available) :] = available
        power = np.abs(np.fft.rfft(window * weights)) ** 2
        fraction = float(power[band].sum() / max(power.sum(), 1e-20))
        if fraction >= OPEN_RATIO:
            opened, low_hops = True, 0
        elif fraction <= CLOSE_RATIO:
            low_hops += 1
            if low_hops >= 2:
                opened = False
        relative_target = float(opened)
        decisions.append(
            {
                "start": start,
                "end": end,
                "target": target,
                "reason": reason,
                "trailing_band_fraction_for_next_packet": fraction,
            }
        )
    return keys, decisions
