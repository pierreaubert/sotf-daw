//! Render actual Dynamic EQ audio for offline setting-assistance experiments.

// Rust guideline compliant 2026-02-21
use serde::Deserialize;
use serde_json::{Value, json};
use sotf_host::test_utils::assert_no_allocs_or_deallocs;
use sotf_host::{CountingAlloc, ParametricInPlacePlugin, ProcessContext};
use sotf_plugin_dynamic_eq::{
    DynamicEqData, DynamicEqPlugin, DynamicEqPluginParams, target_band_response_db,
};
use std::fs::{File, OpenOptions};
use std::io::{BufWriter, Read, Write};
use std::path::PathBuf;

#[global_allocator]
static ALLOCATOR: CountingAlloc = CountingAlloc;
// Offline program and explicit observation-tail bounds; no finite IIR tail claim.
const MAX_SAMPLES: usize = 4_000_000;
const MAX_REQUEST_BYTES: usize = 65_536;
const MAX_ZERO_FRAMES: usize = 65_536;

#[derive(Deserialize)]
#[serde(deny_unknown_fields)]
struct Request {
    input: PathBuf,
    output: PathBuf,
    sample_rate: u32,
    channels: usize,
    frames: usize,
    callback_frames: usize,
    appended_zero_frames: usize,
    response_hz: Vec<f64>,
    params: Value,
}

fn exact_keys(value: &Value, keys: &[&str]) -> Result<(), Box<dyn std::error::Error>> {
    let object = value
        .as_object()
        .ok_or("expected explicit parameter object")?;
    if object.len() != keys.len() || keys.iter().any(|key| !object.contains_key(*key)) {
        return Err("supply the exact complete native parameter schema".into());
    }
    Ok(())
}

fn main() -> Result<(), Box<dyn std::error::Error>> {
    let path = std::env::args_os()
        .nth(1)
        .ok_or("usage: qa-dynamic-eq-render REQUEST.json")?;
    let mut bytes = Vec::new();
    File::open(path)?
        .take((MAX_REQUEST_BYTES + 1) as u64)
        .read_to_end(&mut bytes)?;
    if bytes.len() > MAX_REQUEST_BYTES {
        return Err("request exceeds 64 KiB".into());
    }
    let request: Request = serde_json::from_slice(&bytes)?;
    let samples = request
        .frames
        .checked_mul(request.channels)
        .ok_or("sample count overflow")?;
    if !(44_100..=192_000).contains(&request.sample_rate)
        || !(1..=8).contains(&request.channels)
        || samples == 0
        || samples > MAX_SAMPLES
        || !(1..=4096).contains(&request.callback_frames)
        || request.appended_zero_frames > MAX_ZERO_FRAMES
        || request.response_hz.len() > 1024
        || request
            .response_hz
            .iter()
            .any(|f| !f.is_finite() || *f < 0.0 || *f > f64::from(request.sample_rate) * 0.5)
    {
        return Err("unsupported bounded native render geometry".into());
    }
    exact_keys(
        &request.params,
        &[
            "num_bands",
            "threshold",
            "ratio",
            "attack_ms",
            "release_ms",
            "knee",
            "link_channels",
            "mix",
            "bands",
            "stereo_pairs",
        ],
    )?;
    let bands = request.params["bands"]
        .as_array()
        .ok_or("bands must be an array")?;
    for band in bands {
        exact_keys(
            band,
            &[
                "shape",
                "placement",
                "shelf_slope",
                "frequency",
                "q",
                "gain",
                "band_threshold",
                "band_ratio",
                "active",
                "solo",
            ],
        )?;
    }
    let params: DynamicEqPluginParams = serde_json::from_value(request.params)?;
    if params.bands.len() != params.num_bands {
        return Err("every configured band must be explicit".into());
    }
    // Construct and initialize through the strict actual-rate factory.
    let mut plugin = DynamicEqPlugin::try_from_params_at_sample_rate(
        request.channels,
        params.clone(),
        f64::from(request.sample_rate),
    )?;
    if plugin.latency_samples() != 0 {
        return Err("expected native zero-latency contract".into());
    }
    let mut pcm = Vec::new();
    File::open(&request.input)?
        .take((samples * 4 + 1) as u64)
        .read_to_end(&mut pcm)?;
    if pcm.len() != samples * 4 {
        return Err("PCM must contain exactly frames*channels float32-le samples".into());
    }
    let input: Vec<_> = pcm
        .as_chunks::<4>()
        .0
        .iter()
        .copied()
        .map(f32::from_le_bytes)
        .collect();
    if input.iter().any(|v| !v.is_finite() || v.abs() > 32.0) {
        return Err("PCM must be finite within +/-32".into());
    }
    let frames = request.frames + request.appended_zero_frames;
    let mut output = Vec::with_capacity(frames * request.channels);
    let mut block = vec![0.0; request.callback_frames * request.channels];
    for position in (0..frames).step_by(request.callback_frames) {
        let count = (frames - position).min(request.callback_frames);
        let used = count * request.channels;
        block[..used].fill(0.0);
        let real = request.frames.saturating_sub(position).min(count) * request.channels;
        if real > 0 {
            block[..real].copy_from_slice(
                &input[position * request.channels..position * request.channels + real],
            );
        }
        let context =
            ProcessContext::new(request.sample_rate, count).with_sample_position(position as u64);
        let mut result = Ok(0);
        assert_no_allocs_or_deallocs("Dynamic EQ native fitting render", || {
            result = plugin.process_in_place(&mut block[..used], &context);
        });
        if result? != count {
            return Err("native processor changed accepted frame count".into());
        }
        output.extend_from_slice(&block[..used]);
    }
    if output.len() != frames * request.channels || output.iter().any(|v| !v.is_finite()) {
        return Err("invalid native output".into());
    }
    let responses: Vec<Vec<f64>> = params
        .bands
        .iter()
        .map(|band| {
            request
                .response_hz
                .iter()
                .map(|f| {
                    target_band_response_db(band, *f, f64::from(request.sample_rate))
                        .ok_or("invalid full-target response")
                })
                .collect::<Result<Vec<_>, _>>()
        })
        .collect::<Result<Vec<_>, _>>()?;
    let gr = plugin
        .get_data()
        .and_then(|v| v.downcast::<DynamicEqData>().ok())
        .map(|v| v.gain_reduction_db.as_ref().clone());
    let mut writer = BufWriter::new(
        OpenOptions::new()
            .write(true)
            .create_new(true)
            .open(&request.output)?,
    );
    for sample in output {
        writer.write_all(&sample.to_le_bytes())?;
    }
    writer.flush()?;
    println!(
        "{}",
        json!({"status":"ok", "sample_rate":request.sample_rate,"channels":request.channels,
        "input_frames":request.frames,"output_frames":frames,"appended_zero_frames":request.appended_zero_frames,
        "callback_frames":request.callback_frames,"latency_samples":0,"requested_params":params,
        "response_hz":request.response_hz,"full_target_band_response_db":responses,"last_published_ui_gain_reduction_db":gr,
        "scope":"Actual native Dynamic EQ, fresh construction, no automation; explicit zero extension is processed input and is not a finite IIR tail; static target responses and 10Hz UI GR snapshots are not per-sample applied response or control trajectories"})
    );
    Ok(())
}
