"""Train a small residual mask on native AEC errors from disjoint synthetic scenes.

No microphone recording or pretrained model enters training. Final scene files
are opened only after development selection and frozen weight export. This is
a functional prototype; natural-speech and device-domain acceptance remain open.
"""

from __future__ import annotations

import argparse
import hashlib
import json
import subprocess
import time
from pathlib import Path

import numpy as np
from aec_residual_model import (
    BANDS,
    FEATURES,
    FLOOR,
    FORMAT,
    HIDDEN,
    ResidualModel,
    apply_masks,
)
from scipy import signal

SCENES = (
    "linear",
    "nonlinear",
    "delay_jump",
    "near_only",
    "double_talk",
    "quiet_double_talk",
    "near_after_echo",
    "drift",
)
SPLITS = {
    "train": (20269025, 32),
    "development": (20270025, 16),
    "holdout": (20271025, 16),
}


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def source(rate, frames, rng):
    time_grid = np.arange(frames) / rate
    fundamental = rng.uniform(85, 260) * (
        1 + 0.05 * np.sin(2 * np.pi * rng.uniform(1, 4) * time_grid)
    )
    phase = 2 * np.pi * np.cumsum(fundamental) / rate
    voiced = sum(
        np.sin(harmonic * phase + rng.uniform(0, 2 * np.pi)) / harmonic
        for harmonic in range(1, 10)
    )
    noise = rng.normal(size=frames)
    breath = signal.lfilter([1], [1, -rng.uniform(0.5, 0.9)], noise)
    mixed = 0.65 * voiced + 0.35 * breath
    resonant = np.zeros(frames)
    for center in rng.uniform([250, 900, 2000], [700, 1800, 3500]):
        b, a = signal.iirpeak(center, Q=rng.uniform(2, 5), fs=rate)
        resonant += signal.lfilter(b, a, mixed)
    envelope = np.maximum(
        0, np.sin(2 * np.pi * rng.uniform(1.3, 3.7) * time_grid + rng.uniform(0, 6))
    )
    result = (0.7 * resonant + 0.3 * breath) * envelope
    return result / max(np.sqrt(np.mean(result * result)), 1e-12)


def fixture(rate, scene, seed):
    frames = (6 * rate // 256) * 256
    rng = np.random.default_rng(seed)
    reference = source(rate, frames, rng) * 10 ** (rng.uniform(-28, -14) / 20)
    near_source = source(rate, frames, rng)
    t = np.arange(frames) / rate
    if scene == "near_after_echo":
        reference[t >= 3] = 0
    drive = rng.uniform(2, 7) if scene == "nonlinear" else 0
    playback = np.tanh(drive * reference) / drive if drive else reference.copy()
    position = np.arange(frames, dtype=float)
    delay = np.full(frames, rate * rng.uniform(0.004, 0.012))
    if scene == "delay_jump":
        delay[t >= 3] += rate * 0.012
    if scene == "drift":
        delay += position * rng.choice([-1, 1]) * rng.uniform(50, 150) * 1e-6
    echo = np.zeros(frames)
    offsets = np.r_[0, np.sort(rng.uniform(0.001, 0.025, 7))] * rate
    taps = np.r_[
        rng.uniform(0.35, 0.65),
        rng.normal(size=7) * np.exp(-offsets[1:] / rate / 0.010) * 0.2,
    ]
    for offset, tap in zip(offsets, taps, strict=True):
        echo += tap * np.interp(
            position - delay - offset, position, playback, left=0, right=0
        )
    active = np.zeros(frames, dtype=bool)
    if scene == "near_only":
        echo.fill(0)
        reference.fill(0)
        active[:] = True
    elif scene in ["double_talk", "quiet_double_talk"]:
        active = (t >= 3) & (t < 4.5)
    elif scene == "near_after_echo":
        active = t >= 3
    scale = 0.25 if scene == "quiet_double_talk" else 1.0
    near = near_source * max(np.sqrt(np.mean(echo * echo)), 0.015) * scale * active
    mixture = np.column_stack([echo + near, reference]).astype("<f4")
    return (
        mixture,
        near.astype("<f4"),
        echo.astype("<f4"),
        {
            "scene": scene,
            "seed": seed,
            "rate": rate,
            "frames": frames,
            "playback_drive": drive,
            "tap_offsets": offsets.tolist(),
            "tap_gains": taps.tolist(),
            "source": "generated voiced/noise envelopes; not recorded speech or a physical device capture",
        },
    )


def generate(args):
    if args.output.exists():
        raise ValueError("preserve corpus; choose a new output directory")
    args.output.mkdir(parents=True)
    manifest = {
        "format": "sotf_aec_residual_corpus_v1",
        "renderer_sha256": digest(args.renderer),
        "script_sha256": digest(Path(__file__)),
        "splits": {},
        "source_hashes_disjoint": True,
    }
    seen = set()
    process = subprocess.Popen(
        [str(args.renderer.resolve()), str(args.output.resolve() / "native")],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        text=True,
    )
    try:
        for split, (seed, count) in SPLITS.items():
            rows = []
            for index in range(count):
                rate = [16000, 48000][(index // len(SCENES)) % 2]
                scene = SCENES[index % len(SCENES)]
                micref, near, echo, row = fixture(rate, scene, seed + index)
                identity = f"{split}_{index:03d}"
                directory = args.output / identity
                directory.mkdir()
                for name, array in [("input", micref), ("near", near), ("echo", echo)]:
                    path = directory / (name + ".f32le")
                    path.write_bytes(array.tobytes())
                    row[name + "_sha256"] = digest(path)
                if row["input_sha256"] in seen:
                    raise ValueError("overlapping scenes")
                seen.add(row["input_sha256"])
                request = {
                    "id": identity,
                    "sample_rate": rate,
                    "frames": len(near),
                    "input": str((directory / "input.f32le").resolve()),
                    "near": str((directory / "near.f32le").resolve()),
                }
                process.stdin.write(json.dumps(request) + "\n")
                process.stdin.flush()
                metadata = json.loads(process.stdout.readline())
                if metadata.get("status") != "ok" or metadata["id"] != identity:
                    raise RuntimeError(f"native AEC collection failed: {metadata}")
                row.update(id=identity, native=metadata)
                rows.append(row)
                print(
                    json.dumps(
                        {"split": split, "generated": index + 1, "count": count}
                    ),
                    flush=True,
                )
            manifest["splits"][split] = rows
            (args.output / "manifest.json").write_text(
                json.dumps(manifest, indent=2) + "\n"
            )
    finally:
        process.stdin.close()
        process.wait(timeout=30)
    if process.returncode:
        raise RuntimeError(f"native AEC worker failed: {process.returncode}")


def load_row(row, name):
    path = Path(row["native"]["directory"]) / (name + ".f32le")
    if digest(path) != row["native"][name]["sha256"]:
        raise ValueError("native corpus hash mismatch")
    array = np.fromfile(path, dtype="<f4")
    if len(array) != row["native"][name]["floats"] or not np.isfinite(array).all():
        raise ValueError("native corpus shape/nonfinite error")
    return array


def training_arrays(rows):
    x, y, weights = [], [], []
    for row in rows:
        blocks = row["native"]["blocks"]
        x.append(load_row(row, "features").reshape(blocks * BANDS, FEATURES))
        targets = load_row(row, "targets").reshape(blocks, 2, BANDS)
        y.append(targets[:, 0].reshape(-1))
        weights.append(targets[:, 1].reshape(-1))
    return [np.concatenate(values) for values in [x, y, weights]]


def train(args):
    import torch
    from torch import nn

    if args.output.exists():
        raise ValueError("refuse to overwrite frozen model")
    manifest = json.loads((args.data / "manifest.json").read_text())
    torch.set_num_threads(2)
    torch.manual_seed(20272025)
    x, y, w = [
        torch.from_numpy(a) for a in training_arrays(manifest["splits"]["train"])
    ]
    dx, dy, dw = [
        torch.from_numpy(a) for a in training_arrays(manifest["splits"]["development"])
    ]
    network = nn.Sequential(
        nn.Linear(FEATURES, HIDDEN), nn.Tanh(), nn.Linear(HIDDEN, 1)
    )
    optimizer = torch.optim.AdamW(network.parameters(), lr=0.002, weight_decay=1e-5)
    rng = torch.Generator().manual_seed(20272026)
    best, best_state, trace = float("inf"), None, []
    start = time.perf_counter()
    for epoch in range(args.epochs):
        network.train()
        for indices in torch.randperm(len(x), generator=rng).split(2048):
            optimizer.zero_grad()
            prediction = FLOOR + (1 - FLOOR) * torch.sigmoid(
                network(x[indices]).flatten()
            )
            loss = torch.mean(w[indices] * (prediction - y[indices]) ** 2)
            loss.backward()
            optimizer.step()
        if epoch % 5 == 0 or epoch + 1 == args.epochs:
            with torch.no_grad():
                prediction = FLOOR + (1 - FLOOR) * torch.sigmoid(network(dx).flatten())
                value = float(torch.mean(dw * (prediction - dy) ** 2))
            trace.append({"epoch": epoch, "development_weighted_mse": value})
            if value < best:
                best = value
                best_state = {
                    k: v.detach().clone() for k, v in network.state_dict().items()
                }
            if epoch % 25 == 0:
                print(json.dumps(trace[-1]), flush=True)
    network.load_state_dict(best_state)
    payload = {
        "format": FORMAT,
        "feature_version": 1,
        "bands": BANDS,
        "features": FEATURES,
        "hidden": HIDDEN,
        "gain_floor": FLOOR,
        "w1": network[0].weight.detach().numpy().tolist(),
        "b1": network[0].bias.detach().numpy().tolist(),
        "w2": network[2].weight.detach().numpy().reshape(-1).tolist(),
        "b2": float(network[2].bias.detach().item()),
        "training": {
            "seed": 20272025,
            "epochs": args.epochs,
            "selection": "development_weighted_mse_only",
            "best_development_mse": best,
            "seconds": time.perf_counter() - start,
            "manifest_sha256": digest(args.data / "manifest.json"),
        },
    }
    args.output.mkdir(parents=True)
    path = args.output / "model.json"
    path.write_text(json.dumps(payload, separators=(",", ":"), allow_nan=False) + "\n")
    sha = digest(path)
    (args.output / "model.sha256").write_text(sha + "  model.json\n")
    # Frozen boundary: only now open final scene arrays. Do not choose epochs,
    # thresholds or a replacement checkpoint from their labels or output audio.
    model = ResidualModel(path, sha)
    hx, hy, hw = training_arrays(manifest["splits"]["holdout"])
    reference = model.predict(hx)
    with torch.no_grad():
        native = FLOOR + (1 - FLOOR) * torch.sigmoid(
            network(torch.from_numpy(hx))
        ).numpy().reshape(-1)
    parity = float(np.max(np.abs(reference - native)))
    if parity > 1e-5:
        raise ValueError("NumPy/Torch parity failure")
    report = {
        "model_sha256": sha,
        "weighted_holdout_mse": float(np.mean(hw * (reference - hy) ** 2)),
        "numpy_torch_max_gain_error": parity,
        "development_trace": trace,
        "scope": "native synthetic mask prediction only; no natural speech/device-domain acceptance",
    }
    (args.output / "training-report.json").write_text(
        json.dumps(report, indent=2) + "\n"
    )
    print(
        json.dumps({k: v for k, v in report.items() if k != "development_trace"}),
        flush=True,
    )


def metric(signal_array, error):
    return float(
        10
        * np.log10(max(np.sum(signal_array**2), 1e-30) / max(np.sum(error**2), 1e-30))
    )


def evaluate(args):
    if args.output.exists():
        raise ValueError("preserve evaluation output")
    args.output.mkdir(parents=True)
    manifest = json.loads((args.data / "manifest.json").read_text())
    model = ResidualModel(args.model, args.model_sha256)
    rows = []
    for row in manifest["splits"]["holdout"]:
        blocks = row["native"]["blocks"]
        features = load_row(row, "features").reshape(blocks, BANDS, FEATURES)
        raw = load_row(row, "error_spectra").reshape(blocks, 257, 2)
        error_spectra = raw[:, :, 0].astype(float) + 1j * raw[:, :, 1].astype(float)
        learned, gains = apply_masks(error_spectra, features, model, row["rate"])
        # Cross-check FFT convention before measuring any learned gain.
        adaptive = load_row(row, "adaptive").astype(float)
        reconstruction_error = float(
            np.max(
                np.abs(
                    np.fft.irfft(error_spectra, n=512, axis=1)[:, 256:].reshape(-1)
                    - adaptive
                )
            )
        )
        if reconstruction_error > 1e-6:
            raise ValueError("native error spectrum/time alignment mismatch")
        near = np.fromfile(args.data / row["id"] / "near.f32le", dtype="<f4").astype(
            float
        )
        echo = np.fromfile(args.data / row["id"] / "echo.f32le", dtype="<f4").astype(
            float
        )
        result = {
            "id": row["id"],
            "scene": row["scene"],
            "rate": row["rate"],
            "reconstruction_max_error": reconstruction_error,
            "methods": {},
            "gain_min": float(np.min(gains)),
            "gain_max": float(np.max(gains)),
        }
        for method, audio in [
            ("adaptive", adaptive),
            ("classical", load_row(row, "classical").astype(float)),
            ("neural", learned),
        ]:
            count, rate = row["frames"], row["rate"]
            intervals = {
                "steady": (5 * rate, count),
                "onset": (3 * rate, min(count, 3 * rate + rate // 8)),
            }
            if row["scene"] in ["double_talk", "quiet_double_talk"]:
                intervals["double_talk"] = (13 * rate // 4, 17 * rate // 4)
            if row["scene"] == "near_only":
                intervals = {"near_only": (0, count)}
            if row["scene"] == "near_after_echo":
                intervals["near_after_echo"] = (13 * rate // 4, count)
            scores = {}
            for phase, (first, last) in intervals.items():
                s, e, y = near[first:last], echo[first:last], audio[first:last]
                if np.sum(s * s) > 1e-20:
                    scores[phase] = {
                        "near_snr_db": metric(s, y - s),
                        "near_gain_db": float(
                            20 * np.log10(max(abs(np.dot(y, s) / np.dot(s, s)), 1e-15))
                        ),
                    }
                else:
                    scores[phase] = {"erle_db": metric(e, y)}
            result["methods"][method] = scores
        rows.append(result)
    report = {
        "model_sha256": model.sha256,
        "cases": rows,
        "method": "causal reference inference, not native worker runtime validation",
        "added_model_lookahead": 0,
        "native_aec_latency_samples": 256,
        "automatic_adoption": False,
    }
    (args.output / "report.json").write_text(
        json.dumps(report, indent=2, allow_nan=False) + "\n"
    )
    print(json.dumps({"cases": len(rows), "model_sha256": model.sha256}), flush=True)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    sub = parser.add_subparsers(dest="operation", required=True)
    generate_parser = sub.add_parser("generate")
    generate_parser.add_argument("--renderer", type=Path, required=True)
    generate_parser.add_argument("--output", type=Path, required=True)
    train_parser = sub.add_parser("train")
    train_parser.add_argument("--data", type=Path, required=True)
    train_parser.add_argument("--output", type=Path, required=True)
    train_parser.add_argument("--epochs", type=int, default=200)
    evaluate_parser = sub.add_parser("evaluate")
    evaluate_parser.add_argument("--data", type=Path, required=True)
    evaluate_parser.add_argument("--output", type=Path, required=True)
    evaluate_parser.add_argument("--model", type=Path, required=True)
    evaluate_parser.add_argument("--model-sha256", required=True)
    args = parser.parse_args()
    {"generate": generate, "train": train, "evaluate": evaluate}[args.operation](args)


if __name__ == "__main__":
    main()
