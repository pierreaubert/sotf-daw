#!/usr/bin/env python3
"""Native consequence of explicit noise-only versus speech-contaminated capture."""

import argparse
import hashlib
import json
from pathlib import Path

import numpy as np
from hiss_native import PARAMS, render
from scipy.signal import lfilter


def high_power(audio, start, end, rate):
    alpha = 1 - np.exp(-2 * np.pi * 4000 / rate)
    low = lfilter([alpha], [1, -(1 - alpha)], audio.astype(float), axis=0)
    high = audio.astype(float) - low
    return float(np.mean(high[start:end] ** 2))


def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument("--renderer", type=Path, required=True)
    p.add_argument("--work", type=Path, required=True)
    p.add_argument("--output", type=Path, required=True)
    args = p.parse_args()
    args.work.mkdir(parents=True, exist_ok=False)
    base = Path(
        "crates/sotf-plugins/crates/sotf-plugin-hiss-reducer/tests/fixtures/recorded"
    )
    manifest = json.loads((base / "manifest.json").read_text())
    raw = (base / manifest["file"]).read_bytes()
    if hashlib.sha256(raw).hexdigest() != manifest["pcm_sha256"]:
        raise ValueError("Recorded development fixture hash changed")
    rate = 48000
    speech = (np.frombuffer(raw, dtype="<i2").astype(np.float32) / 32768)[:, None]
    if speech.shape != (rate, 1):
        raise ValueError("Recorded development fixture geometry changed")
    capture = (np.random.default_rng(20691026).normal(size=(rate, 1)) * 0.003).astype(
        np.float32
    )
    contaminated = (capture + speech).astype(np.float32)
    programmes = {
        "noise": (
            np.random.default_rng(20701026).normal(size=(rate * 3, 1)) * 0.003
        ).astype(np.float32),
        "speech": np.tile(speech, (3, 1)),
    }
    captures = {
        "none": None,
        "noise_only": capture,
        "speech_contaminated": contaminated,
    }
    rows = []
    start = int(rate * 1.2)
    end = int(rate * 2.8)
    for spectral in [False, True]:
        for cname, c in captures.items():
            for name, program in programmes.items():
                params = dict(
                    PARAMS,
                    spectral_mode=spectral,
                    use_captured_profile=c is not None,
                    strength=1.0,
                    threshold_db=-60.0,
                )
                work = args.work / f"{spectral}-{cname}-{name}"
                y, meta = render(program, args.renderer, work, params, capture=c)
                before = high_power(program, start, end, rate)
                after = high_power(y, start, end, rate)
                rows.append(
                    {
                        "spectral": spectral,
                        "capture": cname,
                        "programme": name,
                        "metadata": str(work / "metadata.json"),
                        "high_band_attenuation_db": float(
                            10 * np.log10(before / after)
                        ),
                        "full_band_attenuation_db": float(
                            10
                            * np.log10(
                                np.mean(program[start:end].astype(float) ** 2)
                                / np.mean(y[start:end].astype(float) ** 2)
                            )
                        ),
                        "floor_db_per_channel": None
                        if c is None
                        else meta["effective_params"]["captured_profile"][
                            "floor_db_per_channel"
                        ],
                        "source_samples": len(program),
                        "aligned_output_frames": len(y),
                    }
                )
                print(json.dumps(rows[-1]), flush=True)
    result = {
        "scope": "One existing one-speaker sibilance fixture plus generated noise; development consequence measurement only, not a classifier, held-out corpus or population/listening result.",
        "fixture_manifest": str(base / "manifest.json"),
        "fixture_sha256": manifest["pcm_sha256"],
        "region_seconds": [1.2, 2.8],
        "measurement": "Independent exact-mapped one-pole residual at4kHz, measured after declared native delay removal; full public native output retained.",
        "cases": rows,
    }
    args.output.write_text(json.dumps(result, indent=2) + "\n")


if __name__ == "__main__":
    main()
